{# Shared data-grid cell. Link attributes render as target=_blank anchors and attributes flagged with_copy_button get a copy icon — used by the records grid, the view grid and the dashboard table widgets so all three render identically. Args: value = display string, href = link target or none, copy = the attribute's with_copy_button flag. An empty cell renders no copy icon: the grids have nothing to copy, and the display layer reports emptiness as the em dash placeholder (format_value / _display_cell) — the only signal every grid path (base, related and widget columns) carries. #} {%- macro grid_cell(value, href=none, copy=false) -%} {%- set has_value = value is not none and value != '' and value != '—' -%} {%- set safe_href = href | safe_url if href else none -%} {% if safe_href %}{{ value }}{% else %}{{ value }}{% endif %} {% if copy and has_value %} {% endif %} {%- endmacro %}
